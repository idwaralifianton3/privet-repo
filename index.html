import 'moliref.dart'; /// Hasil jawaban Moli. [url] diisi kalau Moli perlu membuka tautan. class Balasan { final String teks; final String? url; const Balasan(this.teks, {this.url}); } bool _mengandung(String s, List kata) => kata.any(s.contains); String _duaDigit(int n) => n.toString().padLeft(2, '0'); /// Padanan dari fungsi jawab(perintah) di Python. Balasan jawab(String perintah) { final p = perintah.toLowerCase(); if (p.contains('jam')) { final now = DateTime.now(); return Balasan('Sekarang jam ${_duaDigit(now.hour)}:${_duaDigit(now.minute)}'); } else if (_mengandung(p, ['siapa kamu', 'kamu siapa', 'who are you'])) { return const Balasan('Saya Moli, asisten pribadi kamu'); } else if (_mengandung( p, ['bersedih', 'sad', 'down', 'diuji', 'banyak masalah', 'sedih'])) { return Balasan(acak(sedih)); } else if (p.contains('motivasi')) { return Balasan(acak(motivasi)); } else if (p.contains('curhat')) { return Balasan(acak(curhat)); } else if (_mengandung(p, ['buka', 'open', 'bukain']) && p.contains('youtube')) { return const Balasan('Oke...', url: 'https://www.youtube.com'); } else if (_mengandung( p, ['lagi apa', 'lagi ngapain', 'what are you doing', 'sedang apa'])) { return Balasan(aktivitas()); } else if (_mengandung(p, ['hai', 'hai,', 'hello', 'moli'])) { return Balasan(acak(jawabanSapaan)); } else { return const Balasan('Maaf, saya belum paham perintah itu'); } }
